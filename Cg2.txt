#include<graphics.h>
#include <conio.h>
#include <math.h>
#include <stdlib.h>
int main()
{
int gd = DETECT, gm;
initgraph(&gd, &gm,(char*) "");
int x1 = 100, y1 = 100;
int x2 = 300, y2 = 200;
int dx = x2 - x1;
int dy = y2 - y1;
int steps = (abs(dx) > abs(dy)) ? abs(dx) : abs(dy);
float xinc = dx / (float)steps;
float yinc = dy / (float)steps;
float x = x1;
float y = y1;
for (int i = 0; i <= steps; i++)
{
putpixel(round(x), round(y), YELLOW);
x += xinc;
y += yinc;
}
outtextxy(150, 80, (char*)"DDA LINE");
getch();
closegraph();
return 0;
}
